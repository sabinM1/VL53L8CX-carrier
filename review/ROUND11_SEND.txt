================================================================================
1. WHAT IS FROZEN
================================================================================

Product: five identical carrier boards, one ST VL53L8CX 8x8 multizone ToF sensor
each, on a Teensy 4.0 (i.MX RT1062, 600 MHz, 3.3 V I/O, not 5 V tolerant).
Sensors run from a separate external 3.3 V rail behind a host-controlled load
switch. The PCB/schematic is not drawn; this phase precedes layout.

Verified external facts used below (sources as cited in earlier rounds):
  DS14161 Rev 7      - SPI <= 3 MHz mode 3, NCS-to-clock >= 0 ns; I2C <= 1 MHz;
                       AVDD 3.13-3.47 V; IOVDD/CORE_1V8 1.62-1.98 V; slew
                       minimums 0.012 V/us (CORE_1V8/IOVDD) and 0.001 V/us
                       (AVDD); 43/50 mA AVDD + 50/80 mA CORE_1V8 active, +10 mA
                       peak each; V_IH >= 0.7 x IOVDD, V_OL <= 0.2 x IOVDD at
                       4 mA; reflow 240 C rec / 260 C max, 3 passes, convection
                       only; all three supplies must come down together.
  SCES898C Rev C     - SN74AXC4T774: OE active-low, referenced to VCCA; 71 kOhm
                       typ static pull-downs on data I/Os (7.3.9); t_EN <= 26 ns,
                       t_DIS <= 20 ns, t_pd <= 7 ns (5.11); external pull-up on a
                       data I/O must be <= 7 kOhm.
  SCES882E           - SN74AXC1T45: DIR = LOW selects the B port as driver (B->A).
                       Pin numbers still unconfirmed.
  TCA9406 Rev G      - Fm+ translator, VCCA 1.65-3.6 V, VCCB 2.3-5.5 V,
                       VCCA <= VCCB, internal 10 kOhm on all four pins, Hi-Z with
                       OE low.
  MIC550x Rev 2.3    - 1.8 V LDO, turn-on 50 us typ / 125 us max, 25 Ohm
                       auto-discharge, V_IH(EN) 1.2 V, 4 MOhm internal EN
                       pull-down.
  TPS22918           - 2 A load switch, R_ON ~52 mOhm typ / <= 78 mOhm, configurable
                       rise time via CT (SR [us/V] = 0.55 x CT[pF] + 30), quick
                       output discharge when off.
  Teensy core 1.62.0 - pin map in 1.3; CNT_SCK_PINS = 1 for every T4.0 SPIClass;
                       Wire BUFFER_LENGTH 136, tx 137.

Items still marked [U] anywhere: SCES882E pin numbers, TPS22918 t_D at CT = 0 /
ON V_IH / current-limit minimum / QOD resistance, MIC5504 exact turn-on endpoints,
the sensor's actual trimmed block set, and re-init behaviour on a streaming
sensor. These are bench-closable and appear in the residual register (section 5).

--- 2.1 Hardware: carrier -----------------------------------------------------

J1 (1x6): 3V3_EXT | GND | SCLK | MOSI | MISO | NCS
   |
   +-- 3V3 --> U2 MIC5504-1.8 (EN = IN, 25 R auto-discharge) --> 1V8
   |            C6 2.2uF in, C7 1uF out
   +-- SCLK/MOSI/MISO --> U3 SN74AXC4T774 (VCCA 1.8 / VCCB 3.3), OE = NCS_S
   +-- NCS --> U5 SN74AXC1T45 (DIR = GND, B->A: host drives sensor),
   |           A = NCS_S, B = NCS_H
   |
   +-- VL53L8CX (SPI mode; AVDD 3.3, CORE_1V8 + IOVDD 1.8)

R1-R5 = 47 kOhm to IOVDD (NCS_S, INT, SYNC, LPn, SPI_I2C_N).
R6   = 6.8 kOhm SCLK_S -> 1V8 (idle 1.8 x 71/77.8 = 1.64 V vs V_IH 1.26 V).
R7   = 4.7 kOhm NCS_H -> 3V3 (hold deselected while host GPIO is a reset input).
C1 = 4.7 uF AVDD; C2/C3/C4/C5/C8/C9 = 100 nF; C7 = 1 uF on 1V8.
20 SMD parts per board, 100 total, plus the system BOM below.

--- 2.2 Hardware: system ------------------------------------------------------

One load switch on 3V3_EXT (TPS22918-class), CT open => t_R ~ 95 us at 3.3 V
(0.0278 V/us, 2.3x the DS14161 minimum). ON pin: 100 kOhm to GND (rail defaults
off), driven by Teensy pin 2. 22 uF at switch VIN, 1 uF at VOUT, >= 47 uF bulk
upstream of the switch. Downstream capacitance 36.5 uF => inrush ~ 1.27 A into a
2 A switch. Rail load 465 mA typ / 750 mA peak; <= 59 mV switch drop.

--- 2.3 Teensy 4.0 pin map ----------------------------------------------------

Bus              Object   SCK/SDA  MOSI  MISO/SCL  CS   Location
SPI (LPSPI4)     SPI      13       11    12        10   header
SPI1 (LPSPI3)    SPI1     27       26    1         0    SCK/MOSI bottom pads
SPI2 (LPSPI1)    SPI2     37       35    34        36   bottom pads
Wire (LPI2C1)    Wire     19       18    -         -    header
Wire1 (LPI2C3)   Wire1    16       17    -         -    header

Pins 0-23 breadboardable; 24-39 bottom pads. Pin 13 = LED (SCK idles high in
mode 3, no alt-SCK). Pins 0/1 = Serial1; 16/17 = Serial4. Pin 2 = rail enable.

--- 2.4 Firmware: mode_a_single_bus.ino (production), verbatim ---------------

~~~cpp
#include <Arduino.h>
#include <SPI.h>
#include <vl53l8cx.h>

#ifndef VL53L8CX_CS_SETTLE_US
#error "Unpatched VL53L8CX library: CS-settle delay missing."
#endif
#if !defined(VL53L8CX_COMMS_CHUNK_SIZE) || VL53L8CX_COMMS_CHUNK_SIZE > 1024
#error "VL53L8CX chunk-size override missing: copy vl53l8cx_platform_config_custom.h"
#endif
#if !defined(VL53L8CX_DISABLE_AMBIENT_PER_SPAD) || !defined(VL53L8CX_DISABLE_SIGNAL_PER_SPAD) \
 || !defined(VL53L8CX_DISABLE_MOTION_INDICATOR)
#error "Output trimming missing: the patched platform_config_custom.h is not being used"
#endif
static_assert(sizeof(VL53L8CX_ResultsData) <= 512,
              "ResultsData should be the trimmed layout (~258 bytes)");

static const uint8_t NUM_SENSORS = 5;
static const uint8_t NCS_PINS[NUM_SENSORS] = {10, 9, 8, 7, 6};
static const uint8_t RAIL_EN_PIN = 2;

static const uint32_t SENSOR_SPI_HZ = 3000000UL;
static const uint8_t  RANGING_HZ    = 15;
static const uint8_t  CENTER_ZONE   = 27;
static const uint8_t  MAX_CONSEC_ERRORS = 50;
static const uint32_t RETRY_PERIOD_MS   = 30000;
static const uint8_t  SILENT_BACKOFF_AFTER = 3;
static const uint32_t SILENT_BACKOFF_MULT  = 10;
static const uint16_t FRAME_PERIOD_MS   = 1000 / RANGING_HZ;
static const uint32_t STALE_MS          = 3UL * FRAME_PERIOD_MS;
static const uint32_t NO_FRAME_TIMEOUT_MS =
  (5UL * FRAME_PERIOD_MS > 2000UL) ? (5UL * FRAME_PERIOD_MS) : 2000UL;

static inline bool status_ok(uint8_t s) { return s == 5 || s == 9; }

static VL53L8CX s0(&SPI, NCS_PINS[0], -1, -1, SENSOR_SPI_HZ);
static VL53L8CX s1(&SPI, NCS_PINS[1], -1, -1, SENSOR_SPI_HZ);
static VL53L8CX s2(&SPI, NCS_PINS[2], -1, -1, SENSOR_SPI_HZ);
static VL53L8CX s3(&SPI, NCS_PINS[3], -1, -1, SENSOR_SPI_HZ);
static VL53L8CX s4(&SPI, NCS_PINS[4], -1, -1, SENSOR_SPI_HZ);
static VL53L8CX *const sensors[NUM_SENSORS] = {&s0, &s1, &s2, &s3, &s4};

static bool     sensor_ok[NUM_SENSORS]     = {false};
static uint8_t  consec_errors[NUM_SENSORS] = {0};
static uint8_t  silent_fails[NUM_SENSORS]  = {0};
static int16_t  last_mm[NUM_SENSORS];
static uint32_t last_update_ms[NUM_SENSORS];
static uint32_t last_retry_ms[NUM_SENSORS];
static VL53L8CX_ResultsData results;

static void park_all_ncs_high() {
  for (uint8_t i = 0; i < NUM_SENSORS; i++) {
    pinMode(NCS_PINS[i], OUTPUT);
    digitalWrite(NCS_PINS[i], HIGH);
  }
}

static bool configure(uint8_t i) {
  uint8_t st = sensors[i]->init();
  if (st != VL53L8CX_STATUS_OK) { Serial.printf("# S%u init FAILED (%u)\n", i, st); return false; }
  st = sensors[i]->set_resolution(VL53L8CX_RESOLUTION_8X8);
  if (st != VL53L8CX_STATUS_OK) { Serial.printf("# S%u set_resolution FAILED (%u)\n", i, st); return false; }
  st = sensors[i]->set_ranging_frequency_hz(RANGING_HZ);
  if (st != VL53L8CX_STATUS_OK) { Serial.printf("# S%u set_frequency FAILED (%u)\n", i, st); return false; }
  st = sensors[i]->set_ranging_mode(VL53L8CX_RANGING_MODE_CONTINUOUS);
  if (st != VL53L8CX_STATUS_OK) { Serial.printf("# S%u set_mode FAILED (%u)\n", i, st); return false; }
  return true;
}

static bool start(uint8_t i) {
  uint8_t st = sensors[i]->start_ranging();
  if (st != VL53L8CX_STATUS_OK) { Serial.printf("# S%u start_ranging FAILED (%u)\n", i, st); return false; }
  return true;
}

static void demote(uint8_t i) {
  sensor_ok[i] = false;
  consec_errors[i] = 0;
  last_mm[i] = -1;
  last_retry_ms[i] = millis();
}

static bool probe(uint8_t i) {
  for (uint8_t a = 0; a < 3; a++) {
    uint8_t alive = 0;
    if (sensors[i]->is_alive(&alive) == VL53L8CX_STATUS_OK && alive) return true;
    delay(50);
  }
  return false;
}

static bool bring_up(uint8_t i) {
  if (!probe(i)) {
    Serial.printf("# S%u is_alive FAILED\n", i);
    return false;
  }
  if (configure(i) && start(i)) {
    consec_errors[i] = 0;
    last_update_ms[i] = millis();
    return true;
  }
  return false;
}

static void bring_up_all() {
  for (uint8_t i = 0; i < NUM_SENSORS; i++) {
    sensor_ok[i] = false;
    if (!probe(i)) {
      Serial.printf("# S%u not responding at boot\n", i);
      continue;
    }
    for (uint8_t a = 0; a < 3 && !sensor_ok[i]; a++) {
      sensor_ok[i] = configure(i);
      if (!sensor_ok[i]) delay(50);
    }
  }
  for (uint8_t i = 0; i < NUM_SENSORS; i++) {
    if (sensor_ok[i]) {
      sensor_ok[i] = start(i);
      if (sensor_ok[i]) {
        last_update_ms[i] = millis();
        Serial.printf("# S%u ranging at %u Hz\n", i, RANGING_HZ);
      }
    }
  }
}

void setup() {
  pinMode(RAIL_EN_PIN, OUTPUT);
  digitalWrite(RAIL_EN_PIN, HIGH);
  delay(5);

  park_all_ncs_high();

  Serial.begin(115200);
  uint32_t t0 = millis();
  while (!Serial && (millis() - t0) < 3000) {}

  for (uint8_t i = 0; i < NUM_SENSORS; i++) {
    last_mm[i] = -1;
    last_update_ms[i] = 0;
    last_retry_ms[i] = 0;
  }

  Serial.println("# VL53L8CX MODE A: single SPI bus (header pins), sequential");

  SPI.begin();
  for (uint8_t i = 0; i < NUM_SENSORS; i++) {
    sensors[i]->begin();
  }
  bring_up_all();

  Serial.println("# ms,S0,S1,S2,S3,S4");
}

void loop() {
  bool updated_any = false;

  for (uint8_t i = 0; i < NUM_SENSORS; i++) {
    if (!sensor_ok[i]) continue;

    uint8_t ready = 0;
    uint8_t st = sensors[i]->check_data_ready(&ready);
    if (st != VL53L8CX_STATUS_OK) {
      if (++consec_errors[i] >= MAX_CONSEC_ERRORS) {
        Serial.printf("# S%u demoted after %u errors\n", i, consec_errors[i]);
        demote(i);
      }
      continue;
    }
    if (!ready) continue;

    if (sensors[i]->get_ranging_data(&results) != VL53L8CX_STATUS_OK) {
      if (++consec_errors[i] >= MAX_CONSEC_ERRORS) {
        Serial.printf("# S%u demoted after %u errors\n", i, consec_errors[i]);
        demote(i);
      }
      continue;
    }

    consec_errors[i] = 0;
    silent_fails[i] = 0;
    const uint8_t z = CENTER_ZONE * VL53L8CX_NB_TARGET_PER_ZONE;
    uint8_t tgt = results.nb_target_detected[CENTER_ZONE];
    uint8_t stt = results.target_status[z];
    last_mm[i] = (tgt > 0 && status_ok(stt)) ? results.distance_mm[z] : -1;
    last_update_ms[i] = millis();
    updated_any = true;
  }

  if (updated_any) {
    uint32_t now = millis();
    Serial.printf("%lu", (unsigned long)now);
    for (uint8_t i = 0; i < NUM_SENSORS; i++) {
      bool fresh = sensor_ok[i] && (now - last_update_ms[i] <= STALE_MS);
      Serial.printf(",%d", fresh ? (int)last_mm[i] : -1);
    }
    Serial.println();
  }

  bool did_retry = false;
  for (uint8_t i = 0; i < NUM_SENSORS; i++) {
    uint32_t period = (silent_fails[i] >= SILENT_BACKOFF_AFTER)
                        ? RETRY_PERIOD_MS * SILENT_BACKOFF_MULT
                        : RETRY_PERIOD_MS;
    if (!sensor_ok[i] && (millis() - last_retry_ms[i] >= period)) {
      last_retry_ms[i] = millis();
      Serial.printf("# retry S%u\n", i);
      sensor_ok[i] = bring_up(i);
      did_retry = true;
      break;
    }
  }

  if (did_retry) {
    uint32_t t = millis();
    for (uint8_t i = 0; i < NUM_SENSORS; i++) {
      if (sensor_ok[i]) last_update_ms[i] = t;
    }
  } else {
    uint32_t now_ms = millis();
    for (uint8_t i = 0; i < NUM_SENSORS; i++) {
      if (sensor_ok[i] && (now_ms - last_update_ms[i] > NO_FRAME_TIMEOUT_MS)) {
        if (silent_fails[i] < SILENT_BACKOFF_AFTER &&
            ++silent_fails[i] == SILENT_BACKOFF_AFTER) {
          Serial.printf("# S%u silent %ux -> retry period %lu ms\n", i,
                        SILENT_BACKOFF_AFTER,
                        (unsigned long)(RETRY_PERIOD_MS * SILENT_BACKOFF_MULT));
        }
        Serial.printf("# S%u no frame for %lu ms -> demoted\n", i,
                      (unsigned long)(now_ms - last_update_ms[i]));
        demote(i);
      }
    }
  }

  delay(2);
}
~~~

3. VERIFICATION MATRIX — fill every row, return the table
================================================================================

Method: A = analysis of the embedded source + cited document
        I = inspection of the embedded text (nothing else needed)
        M = measurement, needs hardware
Status: PASS / FAIL / NV (with the closing action)

--- E: carrier electrical -----------------------------------------------------
ID   Requirement                                              Method Criterion
E1   IOVDD and CORE_1V8 stay within 1.62-1.98 V                A     1.8 V LDO
E2   AVDD stays within 3.13-3.47 V                            A     rail + 59 mV
E3   CORE_1V8 0.9->1.6 V in <= 58.3 us                        M     per board
E4   AVDD to full level in <= 3.3 ms                          M     t_R ~ 95 us
